/// Proverava da li je check-in izvrÅ¡en u dozvoljenom periodu.
bool hasValidCheckIn(Reservation reservation) {
  final checkedInAt = reservation.checkedInAt;

  if (checkedInAt == null) {
    return false;
  }

  return !checkedInAt.isBefore(reservation.checkInOpensAt) &&
      checkedInAt.isBefore(reservation.checkInDeadline);
}

/// OdreÄ‘uje status rezervacije na osnovu trenutnog vremena.
///
/// Funkcija ne menja postojeÄ‡u rezervaciju i ne poziva
/// DateTime.now(), pa je pogodna za unit testiranje.
ReservationStatus resolveReservationStatus(
    Reservation reservation,
    DateTime now,
    ) {
  final status = reservation.status;

  // ZavrÅ¡ena i otkazana rezervacija su konaÄna stanja.
  if (status == ReservationStatus.completed ||
      status == ReservationStatus.cancelled) {
    return status;
  }

  // Rezervacija na Äekanju se otkazuje ako vlasnik
  // nije potvrdio rezervaciju pre isteka roka.
  if (status == ReservationStatus.pending) {
    if (!now.isBefore(reservation.checkInDeadline)) {
      return ReservationStatus.cancelled;
    }

    return ReservationStatus.pending;
  }

  // PotvrÄ‘ena rezervacija bez validnog check-in-a
  // automatski se otkazuje kada istekne rok.
  if (!hasValidCheckIn(reservation) &&
      !now.isBefore(reservation.checkInDeadline)) {
    return ReservationStatus.cancelled;
  }

  // PotvrÄ‘ena rezervacija sa validnim check-in-om
  // automatski se zavrÅ¡ava nakon isteka termina.
  if (hasValidCheckIn(reservation) &&
      !now.isBefore(reservation.endTime)) {
    return ReservationStatus.completed;
  }

  return ReservationStatus.confirmed;
}

/// Proverava da li se Å¾eljeni termin rezervacije
/// preklapa sa nekim blokiranim periodom terena.
///
/// Blokirani periodi drugih terena se ignoriÅ¡u.
bool hasBlockedPeriodOverlap({
  required FieldId fieldId,
  required DateTime startTime,
  required DateTime endTime,
  required Iterable<BlockedPeriod> blockedPeriods,
}) {
  for (final blockedPeriod in blockedPeriods) {
    // Proveravamo samo blokade traÅ¾enog terena.
    if (blockedPeriod.fieldId != fieldId) {
      continue;
    }

    if (doTimeRangesOverlap(
      firstStart: startTime,
      firstEnd: endTime,
      secondStart: blockedPeriod.startTime,
      secondEnd: blockedPeriod.endTime,
    )) {
      return true;
    }
  }

  return false;
}

/// Proverava da li se dva vremenska intervala preklapaju.
///
/// Intervali su [start, end), Å¡to znaÄi da je
/// poÄetak ukljuÄen, a kraj iskljuÄen.
///
/// Primer:
/// 14:00-16:00 i 16:00-18:00 se NE preklapaju.
bool doTimeRangesOverlap({
  required DateTime firstStart,
  required DateTime firstEnd,
  required DateTime secondStart,
  required DateTime secondEnd,
}) {
  return firstStart.isBefore(secondEnd) &&
      secondStart.isBefore(firstEnd);
}

/// Proverava da li postoji konflikt sa postojeÄ‡im
/// aktivnim rezervacijama.
///
/// Konflikt postoji kada:
/// 1. Dve rezervacije koriste isti teren u isto vreme.
/// 2. Isti korisnik ima dve rezervacije u isto vreme,
///    Äak i ako su tereni razliÄiti.
///
/// Otkazane i zavrÅ¡ene rezervacije se ignoriÅ¡u.
bool hasReservationOverlap({
  required FieldId fieldId,
  required UserId userId,
  required DateTime startTime,
  required DateTime endTime,
  required Iterable<Reservation> existingReservations,
}) {
  for (final reservation in existingReservations) {
    if (reservation.status != ReservationStatus.pending &&
        reservation.status != ReservationStatus.confirmed) {
      continue;
    }

    final sameField = reservation.fieldId == fieldId;
    final sameUser = reservation.userId == userId;

    if (!sameField && !sameUser) {
      continue;
    }

    if (doTimeRangesOverlap(
      firstStart: startTime,
      firstEnd: endTime,
      secondStart: reservation.startTime,
      secondEnd: reservation.endTime,
    )) {
      return true;
    }
  }

  return false;
}

/// Pravila dozvoljenih prelaza izmeÄ‘u statusa rezervacije.
///
/// Ova funkcija proverava samo da li je prelaz dozvoljen.
/// Vremenska pravila i check-in proveravamo zasebno.
bool canTransitionReservationStatus(
    ReservationStatus from,
    ReservationStatus to,
    ) {
  return switch (from) {
    ReservationStatus.pending =>
    to == ReservationStatus.confirmed ||
        to == ReservationStatus.cancelled,

    ReservationStatus.confirmed =>
    to == ReservationStatus.completed ||
        to == ReservationStatus.cancelled,

  // ZavrÅ¡ena i otkazana rezervacija su konaÄna stanja.
    ReservationStatus.completed => false,
    ReservationStatus.cancelled => false,
  };
}

/// Proverava da li je vremenski interval rezervacije ispravan.
///
/// null znaÄi da je validacija uspeÅ¡na.
/// U suprotnom vraÄ‡a konkretnu tipiziranu greÅ¡ku.
ReservationValidationError? validateReservationTime({
  required DateTime startTime,
  required DateTime endTime,
  required DateTime now,
}) {
  if (!endTime.isAfter(startTime)) {
    return const InvalidTimeRange();
  }

  if (!startTime.isAfter(now)) {
    return const ReservationInPast();
  }

  return null;
}

/// Proverava da li korisnik sme da izvrÅ¡i check-in.
///
/// Check-in je dozvoljen od 24 sata pre poÄetka
/// do trenutka 5 sati pre poÄetka (iskljuÄivo).
ReservationValidationError? validateCheckIn({
  required Reservation reservation,
  required DateTime now,
}) {
  if (reservation.status != ReservationStatus.confirmed) {
    return const InvalidReservationStatus();
  }

  if (now.isBefore(reservation.checkInOpensAt) ||
      !now.isBefore(reservation.checkInDeadline)) {
    return const CheckInNotAllowed();
  }

  if (reservation.isCheckedIn) {
    return const InvalidReservationStatus();
  }

  return null;
}

/// Proverava da li korisnik sme da otkaÅ¾e rezervaciju.
///
/// Otkazivanje je moguÄ‡e samo pre isteka roka
/// od 5 sati pre poÄetka termina.
ReservationValidationError? validateCancellation({
  required Reservation reservation,
  required DateTime now,
}) {
  if (reservation.status != ReservationStatus.pending &&
      reservation.status != ReservationStatus.confirmed) {
    return const InvalidReservationStatus();
  }

  if (!now.isBefore(reservation.checkInDeadline)) {
    return const CancellationDeadlineExpired();
  }

  return null;
}

/// Proverava da li je Å¾eljeni termin rezervacije
/// u potpunosti unutar radnog vremena terena.
///
/// Za svaki dan koji rezervacija obuhvata proverava
/// odgovarajuÄ‡i raspored rada.
///
/// PodrÅ¾ava termine koji prelaze preko ponoÄ‡i,
/// pod uslovom da je teren otvoren u oba dana.
///
/// Ako raspored za neki dan ne postoji,
/// smatra se da je teren tog dana zatvoren.
bool isReservationWithinWorkingHours({
  required FieldId fieldId,
  required DateTime startTime,
  required DateTime endTime,
  required Iterable<FieldSchedule> schedules,
}) {
  if (!endTime.isAfter(startTime)) {
    return false;
  }

  final fieldSchedules = schedules
      .where((schedule) => schedule.fieldId == fieldId)
      .toList();

  var current = startTime;

  while (current.isBefore(endTime)) {
    final weekday = WeekDay.values[current.weekday - 1];

    FieldSchedule? schedule;

    for (final candidate in fieldSchedules) {
      if (candidate.dayOfWeek == weekday) {
        schedule = candidate;
        break;
      }
    }

    if (schedule == null || !schedule.isOpen) {
      return false;
    }

    final dayStart = DateTime(
      current.year,
      current.month,
      current.day,
    );

    final nextDay = DateTime(
      current.year,
      current.month,
      current.day + 1,
    );

    final segmentEnd = endTime.isBefore(nextDay)
        ? endTime
        : nextDay;

    final openingTime = dayStart.add(
      Duration(minutes: schedule.opensAt.totalMinutes),
    );

    final closingTime = dayStart.add(
      Duration(minutes: schedule.closesAt.totalMinutes),
    );

    // Raspored mora imati pozitivan vremenski interval.
    if (!closingTime.isAfter(openingTime)) {
      return false;
    }

    // Ceo deo rezervacije za ovaj dan mora
    // biti unutar radnog vremena.
    if (current.isBefore(openingTime) ||
        segmentEnd.isAfter(closingTime)) {
      return false;
    }

    current = segmentEnd;
  }

  return true;
}
